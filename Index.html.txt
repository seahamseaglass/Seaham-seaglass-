<script>
const types=[["🤍","White",5],["🤎","Brown",7],["💚","Green",9],["💙","Blue",18],["❤️","Red",45],["🧡","Orange",35],["💛","Yellow",30],["🌈","Multi-colour",60]];
let energy=100,coins=0,level=1,found=0,items=[],started=false;
const game=document.getElementById("game"),player=document.getElementById("player");
function start(){started=true;document.getElementById("intro").style.display="none";spawn();say("Lisa: Find 5 pieces!");}
function say(x){let m=document.getElementById("msg");m.textContent=x;m.style.display="block";clearTimeout(window.tm);window.tm=setTimeout(()=>m.style.display="none",2200)}
function upd(){energy=Math.max(0,energy);document.getElementById("energy").textContent=energy;document.getElementById("coins").textContent=coins;document.getElementById("level").textContent=level}
function spawn(){for(let i=0;i<13;i++){let e=document.createElement("div"),t=types[Math.floor(Math.random()*types.length)];e.className="glass";e.textContent=t[0];e.style.left=(7+Math.random()*86)+"%";e.style.top=(70+Math.random()*25)+"%";e.onclick=()=>collect(e,t);game.appendChild(e);items.push(t)}}
function collect(e,t){if(energy<1)return say("Too tired!");energy--;coins+=t[2];found++;e.remove();upd();say("Found "+t[1]+"! +"+t[2]+" coins");if(found%5===0){level++;say("🎉 Hunt milestone! Level "+level)}}
function search(){if(!started)return;if(energy<3)return say("Not enough energy.");energy-=3;if(Math.random()<.35){let t=types[3+Math.floor(Math.random()*5)];coins+=t[2];found++;say("🔎 You uncovered "+t[1]+" glass! +"+t[2])}else say("Nothing this time.");if(Math.random()<.25)wave();upd()}
function wave(){energy-=15;upd();say("🌊 BIG WAVE! -15 energy")}
function move(dx,dy){if(!started)return;let r=player.getBoundingClientRect(),g=game.getBoundingClientRect();let x=Math.min(g.width-35,Math.max(35,r.left-g.left+dx)),y=Math.min(g.height-65,Math.max(70,r.top-g.top+dy));player.style.left=x+"px";player.style.top=y+"px";if(y>g.height*.68&&Math.random()<.08)wave()}
function bag(){document.getElementById("ptitle").textContent="🎒 Sea-glass Bag";document.getElementById("pbody").innerHTML=items.length?items.map(t=>"🪨 "+t[1]).join("<br>"):"Collect some glass first.";document.getElementById("panel").style.display="block"}
function collection(){document.getElementById("ptitle").textContent="🏆 Collection";document.getElementById("pbody").innerHTML='<div class="grid">'+types.map(t=>'<div class="item"><div style="font-size:30px">'+t[0]+'</div><b>'+t[1]+'</b><br>'+t[2]+' 🪙</div>').join("")+'</div>';document.getElementById("panel").style.display="block"}
function closePanel(){document.getElementById("panel").style.display="none"}
setInterval(()=>{if(started&&Math.random()<.07)wave()},3500);upd();
